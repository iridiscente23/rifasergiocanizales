        document.addEventListener('DOMContentLoaded', () => {
            const grid = document.getElementById('grid');
            
            // Simulación de datos para la demostración visual
            // Puedes cambiar estos arrays manualmente para actualizar tu web luego
            const soldTickets = [12, 45, 88, 7, 99, 23, 67]; 
            const reservedTickets = [5, 50, 14, 82, 33];

            // Generar 100 boletos (00 al 99)
            for (let i = 0; i < 100; i++) {
                const btn = document.createElement('button');
                
                // Formatear a 2 dígitos (00, 01, ..., 99)
                const numberString = i.toString().padStart(2, '0');
                btn.textContent = numberString;
                
                // Determinar el estado del boleto
                if (soldTickets.includes(i)) {
                    btn.classList.add('ticket-btn', 'sold');
                    btn.disabled = true;
                } else if (reservedTickets.includes(i)) {
                    btn.classList.add('ticket-btn', 'reserved');
                    btn.disabled = true;
                } else {
                    btn.classList.add('ticket-btn', 'available');
                    // Agregar evento de click para contactar por WhatsApp
                    btn.addEventListener('click', () => {
                        const numeroWA = 'TUNUMERO'; // PON TU NUMERO AQUÍ (ej: 573100000000)
                        const mensaje = `¡Hola! Me interesa comprar el boleto número *${numberString}* de la rifa. ¿Está disponible?`;
                        const url = `https://wa.me/${numeroWA}?text=${encodeURIComponent(mensaje)}`;
                        window.open(url, '_blank');
                    });
                }
                
                grid.appendChild(btn);
